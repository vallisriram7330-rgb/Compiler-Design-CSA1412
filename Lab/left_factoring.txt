#include <stdio.h>
#include <string.h>

int main()
{
    FILE *in, *out;
    char line[100], lhs;
    char p1[30], p2[30];
    char prefix[30];
    int i, j, k;

    in = fopen("input.txt", "r");
    if (in == NULL)
    {
        printf("Cannot open input.txt\n");
        return 1;
    }

    out = fopen("output.txt", "w");
    if (out == NULL)
    {
        printf("Cannot create output.txt\n");
        fclose(in);
        return 1;
    }

    while (fgets(line, sizeof(line), in))
    {
        line[strcspn(line, "\r\n")] = '\0';

        lhs = line[0];

        if (sscanf(line + 2, "%29[^|]|%29s", p1, p2) != 2)
        {
            fprintf(out, "%s\n", line);
            continue;
        }

        i = 0;

        while (p1[i] != '\0' && p2[i] != '\0' &&
               p1[i] == p2[i])
        {
            prefix[i] = p1[i];
            i++;
        }

        prefix[i] = '\0';

        if (i == 0)
        {
            fprintf(out, "%s\n", line);
        }
        else
        {
            fprintf(out, "%c=%s%c'\n", lhs, prefix, lhs);
            fprintf(out, "%c'=%s|%s\n",
                    lhs, p1 + i, p2 + i);
        }
    }

    fclose(in);
    fclose(out);

    printf("Left factoring completed. Check output.txt\n");

    return 0;
}
