
%{
#include <stdio.h>
#include <string.h>

char word[50];
int count = 0;
%}

%%
[A-Za-z]+ {
    if (strcmp(yytext, word) == 0)
        count++;
}
.|\n    ;
%%

int main()
{
    printf("Enter the word to find frequency: ");
    scanf("%s", word);

    printf("Enter the paragraph:\n");
    yylex();

    printf("\nFrequency of '%s' = %d\n", word, count);
    return 0;
}

int yywrap()
{
    return 1;
}
