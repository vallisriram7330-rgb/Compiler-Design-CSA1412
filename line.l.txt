int main()
{
    FILE *in, *out;

    in = fopen("input_add_line.c", "r");
    if (in == NULL)
    {
        printf("Cannot open input_add_line.c\n");
        return 1;
    }

    out = fopen("output.c", "w");
    if (out == NULL)
    {
        printf("Cannot create output.c\n");
        fclose(in);
        return 1;
    }

    yyin = in;
    yyout = out;

    fprintf(yyout, "1: ");
    yylex();

    fclose(in);
    fclose(out);

    printf("Line numbers added. Check output.c\n");
    return 0;
}
